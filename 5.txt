LR5:

sudo nano /etc/systemd/system/repair-eth.service

[Unit]
Description=Repair Ethernet Service
After=network.target

[Service]
ExecStart=/home/user/repair_eth.sh
Type=oneshot

[Install]
WantedBy=multi-user.target

sudo cmhod 644 /etc/systemd/system/repair-eth.service
sudo systemctl daemon-reload
sudo systemctl enable repair_eth.sh
sudo systemctl status repair_eth.sh

chmod +x /home/user/repair_eth.sh
sudo systemctl start repair_eth.sh
sudo systemctl status repair_eth.sh

sudo reboot

Может работать не у всех — зависит от того, как и куда вы создавали файл. Если у вас не работает, то извините решение искать самим.
