sudo nano /etc/systemd/system/repair-eth.service

[Unit]
Description=Repair Ethernet Service
After=network.target

[Service]
ExecStart=/home/user/repair_eth.sh
Type=oneshot

[Install]
WantedBy=multi-user.target

sudo chmod 644 /etc/systemd/system/repair-eth.service
sudo systemctl daemon-reload
sudo systemctl enable repair-eth.service
sudo systemctl status repair-eth.service

chmod +x /home/user/repair_eth.sh
sudo systemctl start repair-eth.service
sudo systemctl status repair-eth.service

sudo reboot
